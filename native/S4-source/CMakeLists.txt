cmake_minimum_required(VERSION 3.18)
project(S4Optimized LANGUAGES C CXX)
option(S4_REFERENCE "Build original internal math (for comparisons)" OFF)
option(S4_ENABLE_CUDA "Compile optional dynamically-loaded cuBLASXt backend" OFF)
option(S4_NATIVE "Optimize for the build machine CPU" OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
set(S4_SOURCES S4/S4.cpp S4/rcwa.cpp
    S4/fmm/fmm_common.cpp S4/fmm/fmm_FFT.cpp S4/fmm/fmm_kottke.cpp
    S4/fmm/fmm_closed.cpp S4/fmm/fmm_PolBasisNV.cpp S4/fmm/fmm_PolBasisVL.cpp
    S4/fmm/fmm_PolBasisJones.cpp S4/fmm/fmm_experimental.cpp S4/fmm/fft_iface.cpp
    S4/pattern/pattern.c S4/pattern/intersection.c S4/pattern/predicates.c
    S4/numalloc.c S4/gsel.c S4/sort.c S4/kiss_fft/kiss_fft.c
    S4/kiss_fft/tools/kiss_fftnd.c S4/SpectrumSampler.c S4/cubature.c
    S4/Interpolator.c S4/convert.c)
add_library(s4core STATIC ${S4_SOURCES})
target_include_directories(s4core PUBLIC S4 S4/kiss_fft)
# RNP/IO.h collides with Windows <io.h> on case-insensitive filesystems.
target_include_directories(s4core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/S4/RNP>)
if(S4_REFERENCE)
  target_sources(s4core PRIVATE S4/RNP/Eigensystems.cpp)
  set(BLA_SIZEOF_INTEGER 4)
  find_package(BLAS REQUIRED)
  target_link_libraries(s4core PUBLIC ${BLAS_LIBRARIES})
else()
  # The original numerical interfaces use 32-bit Fortran integers: use LP64 only.
  set(BLA_SIZEOF_INTEGER 4)
  find_package(BLAS REQUIRED)
  find_package(LAPACK REQUIRED)
  target_sources(s4core PRIVATE S4/accelerator.cpp)
  target_compile_definitions(s4core PUBLIC HAVE_BLAS HAVE_LAPACK HAVE_S4_ACCEL)
  target_link_libraries(s4core PUBLIC ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ${CMAKE_DL_LIBS})
  if(S4_ENABLE_CUDA)
    find_package(CUDAToolkit REQUIRED)
    if(CUDAToolkit_VERSION VERSION_GREATER_EQUAL 13)
      message(FATAL_ERROR "Use CUDA <=12.x for Quadro P400 (Pascal).")
    endif()
    target_compile_definitions(s4core PRIVATE S4_USE_CUDA)
    target_include_directories(s4core PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  endif()
endif()
if(WIN32)
  target_compile_definitions(s4core PUBLIC _USE_MATH_DEFINES strcasecmp=_stricmp)
endif()
if(S4_NATIVE AND NOT MSVC)
  target_compile_options(s4core PRIVATE -march=native)
endif()
Python3_add_library(S4 MODULE WITH_SOABI S4/main_python.c)
target_link_libraries(S4 PRIVATE s4core)
configure_file(python/s4_parallel.py s4_parallel.py COPYONLY)
configure_file(python/s4_runtime.py s4_runtime.py COPYONLY)
enable_testing()
add_test(NAME parallel COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parallel.py)
if(NOT S4_REFERENCE)
  add_executable(test_gemm tests/test_gemm.cpp)
  target_link_libraries(test_gemm PRIVATE s4core)
  add_test(NAME gemm COMMAND test_gemm)
endif()
